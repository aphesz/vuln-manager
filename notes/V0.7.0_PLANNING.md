#### 1.1 Tiered Matching Strategy

**Tier 1: Exact Matches (100% confidence)**
```python
# backend/app/matching.py

def match_by_cwe(finding: Finding) -> Optional[VulnerabilityTemplate]:
    """Match by CWE ID - highest confidence"""
    if finding.cwe_id:
        return session.query(VulnerabilityTemplate)\
            .filter_by(cwe_id=finding.cwe_id)\
            .first()
    return None

def match_by_cve(finding: Finding) -> Optional[VulnerabilityTemplate]:
    """Match by CVE ID - highest confidence"""
    if finding.cve_id:
        return session.query(VulnerabilityTemplate)\
            .filter_by(cve_id=finding.cve_id)\
            .first()
    return None
```

**Tier 2: Fuzzy String Matching (85-99% confidence)**
```python
from difflib import SequenceMatcher

def calculate_similarity(str1: str, str2: str) -> float:
    """Calculate string similarity using SequenceMatcher"""
    return SequenceMatcher(None, str1.lower(), str2.lower()).ratio()

def match_by_title_fuzzy(finding: Finding, threshold: float = 0.85):
    """Fuzzy match on title"""
    templates = session.query(VulnerabilityTemplate).all()
    best_match = None
    best_score = 0.0
    
    for template in templates:
        score = calculate_similarity(finding.title, template.title)
        if score > best_score and score >= threshold:
            best_score = score
            best_match = template
    
    return best_match, best_score
```

**Tier 3: Semantic Similarity (70-85% confidence) - Optional**
```python
from sentence_transformers import SentenceTransformer

# Load model once at startup
model = SentenceTransformer('all-MiniLM-L6-v2')

def match_by_semantic_similarity(finding: Finding, threshold: float = 0.75):
    """Semantic matching using embeddings"""
    finding_embedding = model.encode(
        f"{finding.title} {finding.description}"
    )
    
    templates = session.query(VulnerabilityTemplate).all()
    best_match = None
    best_score = 0.0
    
    for template in templates:
        template_text = f"{template.title} {template.description}"
        template_embedding = model.encode(template_text)
        
        # Cosine similarity
        score = cosine_similarity(finding_embedding, template_embedding)
        
        if score > best_score and score >= threshold:
            best_score = score
            best_match = template
    
    return best_match, best_score
```

#### 1.2 Matching API Endpoints

# backend/app/main.py

@app.post("/api/findings/{finding_id}/auto-match")
async def auto_match_finding(
    finding_id: int,
    session: Session = Depends(get_session)
):
    """Auto-match finding to best template"""
    finding = session.get(Finding, finding_id)
    if not finding:
        raise HTTPException(404, "Finding not found")
    
    # Try exact matches first
    match = match_by_cwe(finding)
    confidence = 1.0
    method = "cwe"
    
    if not match:
        match = match_by_cve(finding)
        confidence = 1.0
        method = "cve"
    
    # Fall back to fuzzy matching
    if not match:
        match, confidence = match_by_title_fuzzy(finding)
        method = "fuzzy_title"
    
    # Fall back to semantic (if enabled)
    if not match and SEMANTIC_MATCHING_ENABLED:
        match, confidence = match_by_semantic_similarity(finding)
        method = "semantic"
    
    if match:
        # Create match record
        vuln_match = VulnerabilityMatch(
            finding_id=finding.id,
            template_id=match.id,
            confidence_score=confidence,
            match_method=method
        )
        session.add(vuln_match)
        
        # Update finding with template data
        finding.template_id = match.id
        finding.cwe_id = match.cwe_id or finding.cwe_id
        finding.cvss_score = match.cvss_score
        
        session.commit()
        
        return {
            "matched": True,
            "template": match,
            "confidence": confidence,
            "method": method
        }
    
    return {"matched": False}


@app.post("/api/findings/{finding_id}/suggest-matches")
async def suggest_template_matches(
    finding_id: int,
    top_n: int = 5,
    session: Session = Depends(get_session)
):
    """Get top N template suggestions for manual selection"""
    finding = session.get(Finding, finding_id)
    if not finding:
        raise HTTPException(404, "Finding not found")
    
    suggestions = []
    
    # Collect all match types
    templates = session.query(VulnerabilityTemplate).all()
    
    for template in templates:
        scores = []
        
        # CWE exact match
        if finding.cwe_id and finding.cwe_id == template.cwe_id:
            scores.append(("cwe", 1.0))
        
        # Title fuzzy match
        title_sim = calculate_similarity(finding.title, template.title)
        scores.append(("title_fuzzy", title_sim))
        
        # Description fuzzy match
        if finding.description and template.description:
            desc_sim = calculate_similarity(
                finding.description,
                template.description
            )
            scores.append(("description_fuzzy", desc_sim))
        
        # Take best score
        if scores:
            best_method, best_score = max(scores, key=lambda x: x[1])
            if best_score >= 0.5:  # Minimum threshold for suggestions
                suggestions.append({
                    "template": template,
                    "confidence": best_score,
                    "method": best_method
                })
    
    # Sort by confidence and return top N
    suggestions.sort(key=lambda x: x["confidence"], reverse=True)
    return suggestions[:top_n]
```

async def enrich_template_from_nvd(template: VulnerabilityTemplate):

    if not template.cve_id:
        return False
    
    nvd_data = await fetch_cve_details(template.cve_id)
    if not nvd_data:
        return False
    
    # Update template with NVD data

    template.cvss_vector = nvd_data["cvss_v3_vector"]
    template.cvss_score = nvd_data["cvss_v3_score"]
    
    # Extract CWE if not already set
    if not template.cwe_id and nvd_data["cwe_ids"]:
        template.cwe_id = nvd_data["cwe_ids"][0]
    
    template.source = "nvd"
    template.is_verified = True
    template.updated_at = datetime.utcnow()
    
    return True

def create_template_version(
    template: VulnerabilityTemplate,
    changed_by: str,
    reason: str,
    session: Session
):
    """Create a snapshot version before updating template"""
    # Get current max version
    max_version = session.query(
        func.max(VulnerabilityTemplateVersion.version_number)
    ).filter_by(template_id=template.id).scalar() or 0
    
    version = VulnerabilityTemplateVersion(
        template_id=template.id,
        version_number=max_version + 1,
        title=template.title,
        description=template.description,
        cvss_score=template.cvss_score,
        default_risk_rating=template.default_risk_rating,
        changed_by=changed_by,
        change_reason=reason
    )

    return version

@app.post("/api/vulnerability-templates/bulk-create")
async def bulk_create_templates(
    templates: List[VulnerabilityTemplateCreate],
    session: Session = Depends(get_session)
):
    """Create multiple templates at once"""
    created = []
    errors = []
    
    for idx, template_data in enumerate(templates):
        try:
            template = VulnerabilityTemplate(**template_data.dict())
            session.add(template)
            session.flush()
            created.append(template)
        except Exception as e:
            errors.append({"index": idx, "error": str(e)})
    
    session.commit()
    return {
        "created": len(created),
        "errors": errors,
        "templates": created
    }


@app.post("/api/vulnerability-templates/bulk-update")
async def bulk_update_templates(
    updates: List[dict],  # [{"id": 1, "is_verified": true}, ...]
    session: Session = Depends(get_session)
):
    """Update multiple templates at once"""
    updated = []
    
    for update_data in updates:
        template_id = update_data.pop("id")
        template = session.get(VulnerabilityTemplate, template_id)
        if template:
            for key, value in update_data.items():
                setattr(template, key, value)
            updated.append(template)
    
    session.commit()
    return {"updated": len(updated)}


@app.delete("/api/vulnerability-templates/bulk-delete")
async def bulk_delete_templates(
    template_ids: List[int],
    session: Session = Depends(get_session)
):
    """Delete multiple unused templates"""
    deleted = []
    errors = []
    
    for template_id in template_ids:
        template = session.get(VulnerabilityTemplate, template_id)
        if template:
            # Check if used by any findings
            if template.usage_count == 0:
                session.delete(template)
                deleted.append(template_id)
            else:
                errors.append({
                    "id": template_id,
                    "error": "Template in use"
                })
    
    session.commit()
    return {"deleted": len(deleted), "errors": errors}
```

@app.get("/api/vulnerability-templates/analytics")

    total_templates = session.query(VulnerabilityTemplate).count()

## 🚧 Risks & Mitigation

### Technical Risks

**Risk**: Semantic matching too slow for production  
**Impact**: High  
**Probability**: Medium  
**Mitigation**: Make semantic matching optional, pre-compute embeddings, cache results

**Risk**: NVD API rate limits cause delays  
**Impact**: Medium  
**Probability**: High  
**Mitigation**: Implement request queue, use background jobs, cache responses for 24 hours

**Risk**: Versioning increases database size significantly  
**Impact**: Low  
**Probability**: High  
**Mitigation**: Implement version pruning (keep last 10 versions), compress old versions

### User Experience Risks

**Risk**: Too many template suggestions confuse users  
**Impact**: Medium  
**Probability**: Medium  
**Mitigation**: Limit suggestions to top 5, show confidence scores, allow "None of these" option

**Risk**: Bulk operations accidentally delete important templates  
**Impact**: High  
**Probability**: Low  
**Mitigation**: Require confirmation dialog, prevent deletion of in-use templates, implement soft delete

### Project Risks

**Risk**: Scope creep extends timeline  
**Impact**: High  
**Probability**: Medium  
**Mitigation**: Strictly follow phase plan, defer non-critical features to v0.8.0

**Risk**: External API changes break integration  
**Impact**: Medium  
**Probability**: Low  
**Mitigation**: Version API calls, implement fallback behavior, monitor API status
